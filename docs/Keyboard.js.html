/**@module Keyboard */
/**
 * - キーボード入力を管理します
 * @class
 */
export default class Keyboard {
    /**@private @static @type {Map<string, number>} */
    static _keysMap = new Map();

    /**
     * あるキーが押された時間の長さを保持します(ms)
     * @public @static @method
     * @param {string} key 時間を取得したいキー
     * @returns {number}
     */
    static keyPressed(key){
        return this._keysMap.get(key) ?? 0;
    }

    /**
     * キーが押されているかを判定します
     * @public @static @method
     * @param {string} key 判定したいキー
     * @returns {boolean} 推されていればtrue.
     */
    static isPressed(key){
        return this.keyPressed(key) >= 1;
    }

    /**
     * 与えられたキーが全て押されているかを判定します
     * @public @static @method
     * @param  {string[]} keys 判定対象のキー
     * @returns {boolean} 全て押されていればtrue.
     */
    static isAllPressed(...keys){
        return keys.every(key => this.isPressed(key));
    }

    /**
     * キーが押されたときに実行します
     * @public @static @method
     * @param {GlobalEventHandlers} ev 
     */
    static keydown(ev){
        ev.preventDefault();
        const key = this._getKey(ev);
        const prevValue = this._keysMap.get(key) ?? 0;
        this._keysMap.set(key, prevValue + 1);
    }

    /**
     * キーから離されたときに実行します
     * @public @static @method
     * @param {GlobalEventHandlers} ev
     */
    static keyup(ev){
        this._keysMap.delete(this._getKey(ev));
    }

    /**
     * イベントデータから、押されたキーを表す文字列を取得します
     * @private @static @method
     * @param {GlobalEventHandlers} ev 
     * @returns {string}
     */
    static _getKey(ev){
        return ev.key.toLowerCase();
    }
}
